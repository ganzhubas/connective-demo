<script>
(function(){var w=document.querySelector('.cdemo__wrap');if(!w)return;var f=w.querySelector('iframe'),tabs=document.querySelectorAll('.cdemo__tabs button');
/* Демку и заставку грузим только на широком экране, картинку для телефона — только на узком: скрытое не скачивается */
function put(el){if(el&&!el.getAttribute('src'))el.setAttribute('src',el.getAttribute('data-src'))}
function load(){if(window.matchMedia('(min-width:761px)').matches){put(w.querySelector('.cdemo__poster'));put(f)}else put(document.querySelector('.cdemo__mobile'))}
load();window.addEventListener('resize',load);
/* Кнопка мессенджера в углу экрана не перекрывает демку, пока курсор над ней */
w.addEventListener('mouseenter',function(){document.body.classList.add('cdemo-hover')});
function sel(k){tabs.forEach(function(b){b.setAttribute('aria-selected',b.dataset.k===k?'true':'false')})}
window.addEventListener('message',function(e){if(e.source!==f.contentWindow||!e.data)return;var d=e.data;
if(d.msaReady)w.classList.add('ready');if(typeof d.msaSection==='string')sel(d.msaSection);
if(typeof d.msaWheel==='number'&&isFinite(d.msaWheel))window.scrollBy(0,Math.max(-400,Math.min(400,d.msaWheel)));
if(d.msaActive===true)w.classList.add('active')});
setTimeout(function(){w.classList.add('ready')},8000);
tabs.forEach(function(b){b.addEventListener('click',function(){sel(b.dataset.k);f.contentWindow.postMessage({msaNav:b.dataset.k},'*')})});
w.addEventListener('mouseleave',function(){document.body.classList.remove('cdemo-hover');w.classList.remove('active');f.contentWindow.postMessage({msaActive:false},'*')});
})();
</script>
